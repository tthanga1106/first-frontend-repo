// 2. 處理成績數據寫入(POST請求-精準寫入 Google Sheet 9欄)
function doPost(e) {
  try {
    // 解析前端傳送過來的JSON 格式資料
    var params = JSON.parse(e.postData.contents);
    // 取得當前綁定的Google 試算表
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    //依序將9欄資料寫入試算表新列
    sheet.appendRow([
      new Date(),                                                         // A欄: timestamp(寫入時間)
      params.className || "",                                             // B欄: className(班級)
      params.studentId || "",                                             // C欄: studentId(學號)
      params.studentName || "",                                           // D欄: studentName(姓名)
      params.attemptCount || 0,                                           // E欄: attemptCount(嘗試次數)
      params.targetSentence || "",                                        // F欄: targetSentence (練習題目)
      (params.completion !== undefined ? params.completion + "%" : "0%"), // G欄: completion (完整度)
      (params.accuracy !== undefined ? params.accuracy + "%" : "0%"),     // H欄: accuracy(精準度)
      params.transcript || ""                                             // I欄: transcript (語音文本)
    ]);
    // 回傳成功訊息給前端
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
                         .setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    // 發生錯誤時回傳錯誤訊息
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()}))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}
